class Solution {
    Long[][][] dp;

    public long f(int i, int parity, int taken, int[] nums) {
        if (i == nums.length) {
            if (taken == 1 && parity == 1) return Long.MIN_VALUE / 2;
            return 0;
        }

        if (dp[i][parity][taken] != null) {
            return dp[i][parity][taken];
        }

        long sign = (i % 2 == 0) ? 1 : -1;
        long normal = sign * nums[i];
